Quick Code Usage

import render from "json-renderers-specs";
import data from "./data.json" with { type: "json" };
import columns from "./columns.json" with { type: "json" };

// 1. Table
const tableSpec = render({ type: "table.simple", data, columns });

// 2. Form
const formSpec = render({ type: "form", data, columns });

// 3. Select Dropdown
const selectSpec = render({ type: "select", data, columns });

// 4. DataList Autocomplete
const datalistSpec = render({ type: "dataList", data, columns });
